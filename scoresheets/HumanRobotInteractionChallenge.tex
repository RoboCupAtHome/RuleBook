\begin{scorelist}[startbutton=false,timelimit=7]

	\scoreheading{Easy Task --- Social Information Assistant}
	\scoreitem{50}{Correctly detect the request for attention}
	\scoreitem{50}{Correctly identify the requester}
	\scoreitem{100}{Approach the requester appropriately in path, distance, and orientation}
	\scoreitem{50}{Establish an appropriate interaction position}
	\scoreitem{100}{Maintain attention on the requester throughout the interaction}
	\scoreitem{100}{Provide accurate directions to the requested room}
	\scoreitem{25}{Perform an appropriate closing check}
	\scoreitem{25}{Conclude the interaction appropriately}

	\scoreheading{Medium Task --- Social Messenger}
	\scoreitem{25}{Correctly detect and acknowledge the requester}
	\scoreitem{50}{Approach the requester and establish an appropriate interaction position}
	\scoreitem{25}{Recognise that required information is missing}
	\scoreitem{100}{Obtain the missing information through clarification}
	\scoreitem{50}{Maintain attention on the active conversational partner throughout the task}
	\scoreitem{100}{Correctly identify the target guest}
	\scoreitem{50}{Approach the target guest appropriately}
	\scoreitem{75}{Correctly communicate the message or question}
	\scoreitem{75}{Obtain and retain the target guest's response}
	\scoreitem{25}{Return to the original requester}
	\scoreitem{100}{Correctly report the response}
	\scoreitem{25}{Conclude the interaction appropriately}

	\scoreheading{Hard Task --- Social Coordinator}
	\scoreitem{50}{Detect the first guest's request for attention and approach appropriately}
	\scoreitem{100}{Correctly answer the first guest's questions}
	\scoreitem{75}{Correctly detect an interruption or additional participant}
	\scoreitem{75}{Acknowledge the interrupting attendee appropriately}
	\scoreitem{75}{Maintain context of the original interaction}
	\scoreitem{75}{Complete the original interaction before attending to the interrupting attendee}
	\scoreitem{50}{Correctly manage speaker transitions}
	\scoreitem{50}{Direct attention towards the active speaker}
	\scoreitem{75}{Demonstrate appropriate full-duplex conversation behaviour}
	\scoreitem{75}{Maintain correct associations between people, requests, and responses}
	\scoreitem{50}{Return to the waiting attendee without additional prompting}
	\scoreitem{75}{Address the waiting attendee's request without asking them to repeat it}
	\scoreitem{50}{Correctly answer the waiting attendee's request}
	\scoreitem{25}{Conclude both interactions appropriately}

	\scoreheading{Penalties}
	\penaltyitem{50}{Approach a person who did not request assistance}
	\penaltyitem{25}{Approach an incorrect target guest}
	\penaltyitem{25}{Attempt the Medium Task before obtaining the missing information}
	\penaltyitem{25}{Abandon an interaction without appropriate closure}
	\penaltyitem{50}{Fail to return to a waiting attendee after acknowledgement}
	\penaltyitem{25}{Incorrectly associate people, requests, or responses}
	\penaltyitem{25}{Direct responses to the wrong participant during a multi-person interaction}
\end{scorelist}
